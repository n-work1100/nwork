__________________________________________________________________________
R2(config)# ip access-list extended PC5-LIMIT
R2(config-ext-nacl)# permit icmp host 99.99.99.99 10.10.11.0 0.0.0.255 echo
R2(config-ext-nacl)# permit icmp host 99.99.99.99 20.20.21.0 0.0.0.255 echo
R2(config-ext-nacl)# deny  icmp host 99.99.99.99 any
R2(config-ext-nacl)# permit ip any any
R2(config)# interface GigabitEthernet1/0.99
R2(config-subif)# ip access-group PC5-LIMIT in
